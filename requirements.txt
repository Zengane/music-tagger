# torch/torchvision/torchaudio are pinned to the CUDA 13.0 build on purpose.
#
# v1 died here: `muq` depends on x_clip, which depends on torchvision, and pip's
# resolver happily swapped the CUDA torch for a CPU wheel off PyPI. Everything
# still ran, just ~40x slower, on the CPU. Pinning the +cu130 local version
# means pip refuses to substitute a CPU build instead of doing it silently.
#
# macOS has no CUDA: it gets the plain PyPI build, which uses the Apple GPU
# (MPS). PyTorch 2.13 ships macOS wheels for Apple Silicon on macOS 14 or later
# only; Intel Macs are not supported by PyTorch any more.
--extra-index-url https://download.pytorch.org/whl/cu130

torchvision==0.28.0+cu130; sys_platform != "darwin"
torchaudio==2.11.0+cu130; sys_platform != "darwin"
torch==2.13.0; sys_platform == "darwin"
torchvision==0.28.0; sys_platform == "darwin"
torchaudio==2.11.0; sys_platform == "darwin"
